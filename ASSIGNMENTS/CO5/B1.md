#include <iostream>
#include <string>
using namespace std;

class Bill {
protected:
    string patient;

public:
    Bill(string p) : patient(p) {}

    virtual double total() const = 0;
    virtual string type() const = 0;

    void print() const {
        cout << type() << " " << patient << ": " << total() << endl;
    }

    virtual ~Bill() {}
};

class OutPatient : public Bill {
    double consultation, tests;

public:
    OutPatient(string p, double c, double t)
        : Bill(p), consultation(c), tests(t) {}

    double total() const override {
        return consultation + tests;
    }

    string type() const override {
        return "OP";
    }
};

class InPatient : public Bill {
    int days;
    double rate, surgery;

public:
    InPatient(string p, int d, double r, double s)
        : Bill(p), days(d), rate(r), surgery(s) {}

    double total() const override {
        return days * rate + surgery;
    }

    string type() const override {
        return "IP";
    }
};

int main() {
    Bill* b1 = new OutPatient("Ravi", 500, 1200);
    Bill* b2 = new InPatient("Meena", 3, 2500, 40000);

    b1->print();
    b2->print();

    double sum = b1->total() + b2->total();
    cout << "Total " << sum << endl;

    delete b1;
    delete b2;
